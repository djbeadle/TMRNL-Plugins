{% template map_javascript %}
<script>
  // Initialize map
  let map = L.map('map', {
    zoomControl: false,
    scrollWheelZoom: false,
    doubleClickZoom: false,
    boxZoom: true,
    keyboard: false,
    dragging: true,
    maxZoom: 10,
    attributionControl: false
  }).setView([0, 0], 3);

  // Add CartoDB Positron tiles (clean without text labels)
  L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
    attribution: 'CartoDB',
    subdomains: 'abcd',
    maxZoom: 10,
    minZoom: 2,
    className: 'image-dither',
  }).addTo(map);


  function displayMap() {
    const lat = "{{trmnl.plugin_settings.custom_fields_values.latitude}}";
    const lng = "{{trmnl.plugin_settings.custom_fields_values.longitude}}";

    map.fitBounds([
      [{{bbox[1]}}, {{bbox[0]}}],
      [{{bbox[4]}}, {{bbox[3]}}]
    ], {padding: [{{trmnl.plugin_settings.custom_fields_values.additionalPadding | plus: 20}},{{ trmnl.plugin_settings.custom_fields_values.additionalPadding | plus: 20 }}]});
                  }

                  {% for eq in features %}
                  {% if trmnl.plugin_settings.custom_fields_values.showAdditionalEarthquakes != "yes" and forloop.index > eq_limit  %}
                  /* SKIPPING! */
                  {% continue %}
                  {% endif %}
                  L.marker([{{eq.geometry.coordinates[1]}}, {{eq.geometry.coordinates[0]}}], {
                    icon: L.BeautifyIcon.icon({
                      {% if forloop.index <= eq_limit %} icon: "{{ forloop.index }}", iconShape: "marker" {% else %} iconShape: "rectangle-dot" {% endif %}
                                                                                                               }),
                  draggable: false
                  }).setZIndexOffset({{ eq_limit | minus: forloop.index | times: 10 }}).addTo(map);
  {% endfor %}             

  // Render the map
  displayMap();
</script>
{% endtemplate %}
